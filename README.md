# dir-index.sh action

Use [dir-index.sh] to generate static index files.

[dir-index.sh]: https://github.com/jefferyto/dir-index.sh

## Usage

```yaml
- uses: jefferyto/dir-index.sh-action@v1
  with:
    # General options:

    # The branch, tag or SHA to use for dir-index.sh
    # Default: "1.3.0"
    dir-index-sh-ref: ""

    # Relative paths (comma and/or newline separated) under
    # $GITHUB_WORKSPACE to generate directory index files
    # Default: "."
    paths: ""

    # Whether to overwrite existing directory index files
    # Valid values: "true" or "false"
    # Default/empty value: "false"
    overwrite: ""


    # Indexing options:

    # Ignore files and directories matching these (comma and/or newline
    # separated) shell patterns when indexing a directory
    ignore-patterns: ""

    # Whether to ignore dot files and directories (.*)
    # Valid values: "true" or "false"
    # Default/empty value: "true"
    ignore-dot: ""

    # Whether to ignore base name files included with `*-include`
    # options (see HTML-specific output options)
    # Valid values: "true" or "false"
    # Default/empty value: "true"
    ignore-includes: ""

    # Skip directories matching these (comma and/or newline separated)
    # shell patterns and their subdirectories when searching for
    # directories to index
    skip-patterns: ""

    # Whether to skip dot directories (.*)
    # Valid values: "true" or "false"
    # Default/empty value: "true"
    skip-dot: ""

    # Whether to apply ignore patterns to skip directories
    # Valid values: "true" or "false"
    # Default/empty value: "false"
    skip-ignores: ""

    # Ignore patterns and skip patterns match base names only. If a path
    # is given, all leading directories will be removed from the
    # pattern.


    # Output options:

    # File name to use for directory index files
    # Default/empty value: index.html, index.xml or index.json
    index-file-name: ""

    # Format to use for directory index files
    # Valid values: "html", "xml" or "json"
    # Default/empty value: "html"
    index-file-format: ""

    # File property to sort directory index by
    # Valid values: "name", "modified" or "size"
    # Default/empty value: "name"
    sort-by: ""

    # Whether to sort ascending or descending
    # Valid values: "ascending"/"asc" or "descending"/"desc"
    # Default/empty value: "ascending"
    sort-direction: ""

    # Whether to sort by version name and number ("true") or by
    # lexicographical order ("false")
    # Valid values: "true" or "false"
    # Default/empty value: "true"
    version-sort: ""

    # Whether to perform case-insensitive sort
    # Valid values: "true" or "false"
    # Default/empty value: "false"
    case-insensitive: ""

    # Whether to list directories before files ("true") or list files
    # and directories together ("false")
    # Valid values: "true" or "false"
    # Default/empty value: "false"
    folders-first: ""

    # Whether to update last modified dates of directories when creating
    # directory index files
    # Valid values: "true" or "false"
    # Default/empty value: "false"
    update-directory-last-modified: ""


    # HTML-specific output options:

    # Built-in style to use
    # Valid values: "none", "classic" or "classic-full"
    # Default/empty value: "none"
    style: ""

    # Text to use for page title (<title>)
    # Use "%s" for the current index path, "%%s" for a literal "%s"
    # Default/empty value: "Index of %s"
    page-title: ""

    # Text to use for page text direction (<html dir="...">)
    # Default/empty value: none
    page-text-direction: ""

    # Text to use for page language (<html lang="...">)
    # Default/empty value: "en"
    page-language: ""

    # Text to use for page charset (<meta charset="...">)
    # Default/empty value: "utf-8"
    page-charset: ""

    # HTML to include in <head>
    head-html: ""

    # HTML to use for header
    # Use "%s" for the current index path, "%%s" for a literal "%s"
    # Default/empty value: "Index of %s"
    header-html: ""

    # HTML to use for breadcrumbs root link
    # Default/empty value: "Home"
    breadcrumbs-root-link-html: ""

    # HTML to use for readme
    # Default/empty value: "Generated by dir-index.sh <version>"
    readme-html: ""

    # File to include in <head>
    head-include: ""

    # File to include in place of header
    header-include: ""

    # File to include in place of readme
    readme-include: ""

    # For `head-include`, `header-include` and `readme-include`:
    #   * Option value is a base name: for each directory to be indexed,
    #     if a file with that name exists in the directory then it will
    #     be included in that directory's index file
    #   * Option value is a path: the file at that path (relative to
    #     $GITHUB_WORKSPACE) will be included in every directory index
    #     file

    # Comma-separated list of columns to include in directory listing
    # Valid values: "name", "modified" and/or "size"
    # Default/empty value: "name, modified, size"
    columns: ""

    # HTML to use for file name column heading
    # Default/empty value: "Name"
    name-column-heading: ""

    # Format string for file last modified dates
    # Same syntax as 'date' command output format string
    # Default/empty value: "%Y-%m-%d %H:%M"
    last-modified-format: ""

    # Whether to output file last modified dates in UTC
    # Valid values: "true" or "false"
    # Default/empty value: "false"
    last-modified-utc: ""

    # HTML to use for file last modified column heading
    # Default/empty value: "Last modified"
    last-modified-column-header: ""

    # Unit for file sizes
    # Valid values:
    #   * "none" (exact size)
    #   * "si" (1k = 1000, 1M = 1000000, ...)
    #   * "iec" (1K = 1024, 1M = 1048576, ...)
    #   * "iec-i" (1Ki = 1024, 1Mi = 1048576, ...)
    # Default/empty value: "iec"
    size-unit: ""

    # HTML to use for file size column heading
    # Default/empty value: "Size"
    size-column-heading: ""

    # Text to use for parent directory link
    # Default/empty value: "Parent Directory"
    parent-directory-link-text: ""

    # Generate directory index files as under this path instead of at
    # web server root
    # Should be an absolute path
    index-path: ""

    # Whether to output directory index as a table ("false") or as a
    # list of file names only ("true")
    # Valid values: "true" or "false"
    # Default/empty value: "false"
    output-as-list: ""

    # Whether to output horizontal rule lines (<hr>) in directory index
    # table
    # Valid values: "true" or "false"
    # Default/empty value: "true"
    horizontal-rule-lines: ""

    # Whether to include opening and closing HTML ("true") or (with
    # "false"):
    #   * omit opening HTML (<html>, <head>, etc.) when including file
    #     content with `header-include`, and
    #   * omit closing HTML (</body></html>) when including file content
    #     with `readme-include`
    # Valid values: "true" or "false"
    # Default/empty value: "true"
    html-preamble: ""


    # XML-specific output options:

    # Encoding to specify in XML declaration (<?xml encoding="..."?>)
    # Default/empty value: "UTF-8"
    xml-encoding: ""
```
